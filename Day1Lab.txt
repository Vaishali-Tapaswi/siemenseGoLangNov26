Lab10 - function as return (closures)
package main

import "fmt"

func process(base int) func(int) int{
	fmt.Println("in process function with base = ", base)
	return func(a int) int {
		fmt.Println("in return function")
		base = a+base
		return base
	}
}	

func main() {
	fp := process(5)
	fmt.Println("Result of fp is:", fp(10))
	fmt.Println("Result of fp is:", fp(10))
	fmt.Println("Result of fp is:", fp(10))
}

Lab3 - functions (Lab3.go) 
package main

import "fmt"
func helloworld(){
	fmt.Println("in Hello World function")
}
func add(a , b int) int {
	fmt.Println("in add function")
	return a+b
}
func sub(a , b int) int {
	fmt.Println("in sub function")
	return a-b
}
func calc(a , b int) (int, int) {
	fmt.Println("in calc function")
	return a+b, a-b
}
func main() {
	helloworld()
	result := add(5, 10)
	fmt.Println("Result of addition is:", result)
	result = sub(10, 5)
	fmt.Println("Result of subtraction is:", result)
	addResult, subResult := calc(10, 5)
	fmt.Println("Returned, sum =  ", addResult, "sub = ", subResult)
}
