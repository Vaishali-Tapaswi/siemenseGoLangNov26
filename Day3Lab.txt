    "database/sql"

    "log"

)

type Emp struct {
	Empno   int     `json:"empno"`
	Empname string  `json:"empname"`
	Sal  float64 `json:"salary"`
}

func getEmployeeById(db *sql.DB, empno int) {
    var e Emp
    err := db.QueryRow("select * from emptable where empno=$1", empno).Scan(&e.Empno, &e.Empname, &e.Sal)
    if err != nil {
        log.Fatal(err)
    }
    log.Println("Retrieved Employee :", e.Empno, " Employee name :", e.Empname, " Salary :", e.Sal)
}
 
func Update(db *sql.DB, emp Emp) {
    result, err := db.Exec("update emptable set salary=$1, ename=$2 where empno=$3", emp.Sal, emp.Empname, emp.Empno)
    log.Println("result :", result, "error :", err)
    rowsaff, err := result.RowsAffected()
    if err != nil {
        log.Fatal(err)
        return
    }
    log.Println("rowsaffect  : ", rowsaff)
}
 
func Delete(db *sql.DB, empno int) {
    result, err := db.Exec("DELETE FROM emptable WHERE empno = $1", empno)
    if err != nil {
        fmt.Println("delete failed:", err)
        return
    }
 
    rowsAffected, err := result.RowsAffected()
    if err != nil {
        fmt.Println("could not get deleted row count:", err)
        return
    }
    fmt.Println("rows deleted:", rowsAffected)
}

    rows, err := db.Query("Select empno, ename,salary from emptable")
	if err != nil {
		fmt.Println("select error" , err)

	defer rows.Close()
    var emps []Emp
    for rows.Next() {
        var emp Emp
        // check error
        rows.Scan(&emp.Empno, &emp.Empname, &emp.Sal)
        emps = append(emps, emp)

    return emps
}

package main

import (
	"fmt"
	"time"
	//"math/rand"
	"sync"
)

type Bank struct {
	mutex sync.Mutex
	Balance int 
}

func  deposit(bank Bank) {
	for i:=0;i<100;i++{
		// chances of switching go routines become highter
		bank.mutex.Lock()
		//defer bank.mutex.Unlock()
		bal := bank.Balance
		bal += 1
		//time.Sleep(time.Duration(rand.Intn(10)) * time.Millisecond)
		bank.Balance = bal	
		bank.mutex.Unlock()
	}
	fmt.Println("In Deposit - " ,  bank.Balance )
}


func  withdraw(bank Bank) {
	for i:=0;i<100;i++{
		// chances of switching go routines become highter
		bank.mutex.Lock()
		bal := bank.Balance
		bal -= 1
		//time.Sleep(time.Duration(rand.Intn(10)) * time.Millisecond)
		bank.Balance = bal	
		bank.mutex.Unlock()
	}
	fmt.Println("In withdraw - " ,  bank.Balance )
}

func main() {
	bank := Bank{}
   
    for i := 0; i < 5; i++ {
       go deposit(bank)
       go withdraw(bank)
    }

    time.Sleep(2000 * time.Millisecond)
    fmt.Println("Main - Final balance :", bank.Balance)
}
